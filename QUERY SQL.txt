--Sales Date Prediction


WITH OrderIntervals AS (
    SELECT 
        C.custid,
        C.companyname AS CustomerName,
        O.orderdate,
        LAG(O.orderdate) OVER (PARTITION BY C.custid ORDER BY O.orderdate) AS PrevOrderDate,
        DATEDIFF(DAY, 
                 LAG(O.orderdate) OVER (PARTITION BY C.custid ORDER BY O.orderdate), 
                 O.orderdate) AS DaysBetweenOrders
    FROM Sales.Customers C
    INNER JOIN Sales.Orders O ON C.custid = O.custid
)
, CustomerPredictions AS (
    SELECT 
        custid,
        CustomerName,
        MAX(orderdate) AS LastOrderDate,
        AVG(CAST(DaysBetweenOrders AS FLOAT)) AS AvgDaysBetweenOrders,
        COUNT(DaysBetweenOrders) AS NumberOfIntervals
    FROM OrderIntervals
    WHERE PrevOrderDate IS NOT NULL
    GROUP BY custid, CustomerName
)
SELECT 
	custid,
    CustomerName,
    LastOrderDate,
    DATEADD(DAY, ROUND(AvgDaysBetweenOrders, 0), LastOrderDate) AS NextPredictedOrder
FROM CustomerPredictions
WHERE NumberOfIntervals > 0
ORDER BY CustomerName;

GO
SET QUOTED_IDENTIFIER ON

CREATE PROCEDURE [Sales].[usp_AddNewOrder]
    @CustId INT,                 -- Id del cliente
    @EmpId INT,                  -- Id del empleado
    @OrderDate DATETIME,
    @RequiredDate DATETIME,
    @ShippedDate DATETIME = NULL,
    @ShipperId INT,
    @Freight MONEY,
    @ShipName NVARCHAR(100),
    @ShipAddress NVARCHAR(200),
    @ShipCity NVARCHAR(100),
    @ShipCountry NVARCHAR(100),

  
    @ProductId INT,
    @UnitPrice MONEY,
    @Qty SMALLINT,
    @Discount REAL = 0.0,

    @NewOrderId INT OUTPUT 
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        -- 1. Insertar en Orders
        INSERT INTO Sales.Orders (
            custid,
            empid,
            orderdate,
            requireddate,
            shippeddate,
            shipperid,
            freight,
            shipname,
            shipaddress,
            shipcity,
            shipcountry
        )
        VALUES (
            @CustId,
            @EmpId,
            @OrderDate,
            @RequiredDate,
            @ShippedDate,
            @ShipperId,
            @Freight,
            @ShipName,
            @ShipAddress,
            @ShipCity,
            @ShipCountry
        );

        -- Obtener el ID de la orden creada
        SET @NewOrderId = SCOPE_IDENTITY();

        -- 2. Insertar en OrderDetails
        INSERT INTO Sales.OrderDetails (
            orderid,
            productid,
            unitprice,
            qty,
            discount
        )
        VALUES (
            @NewOrderId,
            @ProductId,
            @UnitPrice,
            @Qty,
            @Discount
        );

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH
END;
